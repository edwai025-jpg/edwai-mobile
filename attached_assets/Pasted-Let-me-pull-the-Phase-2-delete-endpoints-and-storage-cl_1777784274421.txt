Complete Mobile Delete Flow

const API_BASE = 'https://edwai-api-953672603784.asia-south1.run.app';
async function deleteDocument(documentId, token) {
  const response = await fetch(
    `${API_BASE}/api/phase2/documents/${documentId}`,
    {
      method: 'DELETE',
      headers: {
        Authorization: `Bearer ${token}`,
      },
    }
  );
  if (response.status === 404) {
    throw new Error('Document not found');
  }
  if (response.status === 400) {
    const body = await response.json();
    throw new Error(body.detail); // "Cannot delete a document that is currently being processed"
  }
  if (!response.ok) {
    throw new Error('Delete failed. Please try again.');
  }
  const result = await response.json();
  // result.status === "deleted" | "deleted_with_warnings"
  // Both mean: remove from local state, update your list
  return result;
}
